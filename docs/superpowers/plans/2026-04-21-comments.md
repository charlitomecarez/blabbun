# Comments Section Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an anonymous comment section with one-level replies, mock data, and in-memory state to every blog post detail page.

**Architecture:** Three new files — `types.ts` for shared interfaces, `CommentItem.svelte` (leaf: renders one comment + replies + inline reply form), `Comments.svelte` (root: owns mock data, list, add-comment form). `[slug].astro` is modified to mount the island. Supabase integration later only requires changing `Comments.svelte`.

**Tech Stack:** Svelte 5 (runes), Tailwind CSS, Astro `client:load` island, `crypto.randomUUID()`

<script lang="ts">
  import type { Comment } from "./types";

  interface Props {
    comment: Comment;
    activeReplyId: string | null;
    onReplyToggle: (id: string | null) => void;
    onReplySubmit: (commentId: string, body: string) => void;
  }

  const { comment, activeReplyId, onReplyToggle, onReplySubmit }: Props =
    $props();

  let replyBody = $state("");

  const isReplying = $derived(activeReplyId === comment.id);

  function relativeTime(date: Date): string {
    const seconds = Math.floor((Date.now() - date.getTime()) / 1000);
    if (seconds < 60) return "just now";
    const minutes = Math.floor(seconds / 60);
    if (minutes < 60) return `${minutes} minute${minutes === 1 ? "" : "s"} ago`;
    const hours = Math.floor(minutes / 60);
    if (hours < 24) return `${hours} hour${hours === 1 ? "" : "s"} ago`;
    const days = Math.floor(hours / 24);
    if (days < 7) return `${days} day${days === 1 ? "" : "s"} ago`;
    const weeks = Math.floor(days / 7);
    return `${weeks} week${weeks === 1 ? "" : "s"} ago`;
  }

  function handleReplySubmit() {
    if (!replyBody.trim()) return;
    onReplySubmit(comment.id, replyBody.trim());
    replyBody = "";
  }

  function handleCancel() {
    replyBody = "";
    onReplyToggle(null);
  }
</script>

<div>
  <!-- Comment header -->
  <div class="mb-1.5 flex items-baseline gap-2">
    <span class="font-manrope text-[13px] font-bold text-[#29302d]"
      >Anonymous</span
    >
    <span class="font-manrope text-[11px] text-[rgba(41,48,45,0.4)]"
      >{relativeTime(comment.createdAt)}</span
    >
  </div>

  <!-- Comment body -->
  <p
    class="font-manrope mb-2 text-[15px] leading-[1.65] text-[rgba(41,48,45,0.8)]"
  >
    {comment.body}
  </p>

  <!-- Reply toggle button -->
  <button
    type="button"
    onclick={() => onReplyToggle(isReplying ? null : comment.id)}
    class="font-manrope cursor-pointer border-none bg-transparent p-0 text-[12px] font-bold tracking-[0.04em] text-[#386547]"
  >
    ↩ Reply
  </button>

  <!-- Replies + inline reply form -->
  {#if comment.replies.length > 0 || isReplying}
    <div class="mt-3 ml-5 space-y-3 border-l-2 border-[#c5f7d0] pl-4">
      {#each comment.replies as reply (reply.id)}
        <div>
          <div class="mb-1 flex items-baseline gap-2">
            <span class="font-manrope text-[13px] font-bold text-[#29302d]"
              >Anonymous</span
            >
            <span class="font-manrope text-[11px] text-[rgba(41,48,45,0.4)]"
              >{relativeTime(reply.createdAt)}</span
            >
          </div>
          <p
            class="font-manrope text-[14px] leading-[1.65] text-[rgba(41,48,45,0.75)]"
          >
            {reply.body}
          </p>
        </div>
      {/each}

      {#if isReplying}
        <div>
          <textarea
            bind:value={replyBody}
            placeholder="Write a reply…"
            rows="3"
            class="font-manrope w-full resize-none rounded-[8px] border-[1.5px] border-[rgba(197,247,208,0.8)] bg-white px-3 py-2.5 text-[14px] text-[#29302d] placeholder-[rgba(41,48,45,0.35)] outline-none focus:border-[#386547]"
          ></textarea>
          <div class="mt-2 flex gap-2">
            <button
              type="button"
              onclick={handleReplySubmit}
              disabled={!replyBody.trim()}
              class="font-manrope cursor-pointer rounded-full bg-[#386547] px-4 py-1.5 text-[13px] font-bold text-white disabled:cursor-not-allowed disabled:opacity-40"
            >
              Post reply
            </button>
            <button
              type="button"
              onclick={handleCancel}
              class="font-manrope cursor-pointer border-none bg-transparent text-[13px] font-bold text-[rgba(41,48,45,0.4)] hover:text-[rgba(41,48,45,0.7)]"
            >
              Cancel
            </button>
          </div>
        </div>
      {/if}
    </div>
  {/if}
</div>

### Task 3: Create `Comments.svelte`

**Files:**

- Create: `apps/client/src/components/post/Comments.svelte`

- [ ] **Step 1: Create the file**

```svelte
<script lang="ts">
  import type { Comment } from "./types";
  import CommentItem from "./CommentItem.svelte";

  interface Props {
    postSlug: string;
  }

  // postSlug will be the Supabase query key when backend is integrated
  const { postSlug: _postSlug }: Props = $props();

  let comments = $state<Comment[]>([
    {
      id: "c1",
      body: "This really resonated with me. The part about letting thoughts settle without forcing them to be complete — that's something I needed to read today.",
      createdAt: new Date(Date.now() - 2 * 24 * 60 * 60 * 1000),
      replies: [
        {
          id: "r1",
          body: "Same here. Bookmarked this one.",
          createdAt: new Date(Date.now() - 1 * 24 * 60 * 60 * 1000),
        },
      ],
    },
    {
      id: "c2",
      body: "Beautifully written. Blab Bun always finds a way to put words to things I couldn't articulate.",
      createdAt: new Date(Date.now() - 5 * 24 * 60 * 60 * 1000),
      replies: [],
    },
    {
      id: "c3",
      body: "Found this through a friend. Glad I did.",
      createdAt: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000),
      replies: [],
    },
  ]);

  let activeReplyId = $state<string | null>(null);
  let newCommentBody = $state("");

  function addComment() {
    if (!newCommentBody.trim()) return;
    comments = [
      {
        id: crypto.randomUUID(),
        body: newCommentBody.trim(),
        createdAt: new Date(),
        replies: [],
      },
      ...comments,
    ];
    newCommentBody = "";
  }

  function addReply(commentId: string, body: string) {
    comments = comments.map((c) =>
      c.id === commentId
        ? {
            ...c,
            replies: [
              ...c.replies,
              { id: crypto.randomUUID(), body, createdAt: new Date() },
            ],
          }
        : c
    );
    activeReplyId = null;
  }
</script>

<div
  class="mx-auto max-w-[760px] border-t-2 border-[#c5f7d0] px-4 pt-10 pb-12 md:px-8"
>
  <!-- Section header -->
  <p
    class="font-manrope mb-6 text-[11px] font-bold tracking-[0.2em] text-[rgba(41,48,45,0.5)] uppercase"
  >
    {comments.length} Comment{comments.length === 1 ? "" : "s"}
  </p>

  <!-- Comment list -->
  {#if comments.length > 0}
    <div class="space-y-6">
      {#each comments as comment, i (comment.id)}
        <div>
          <CommentItem
            {comment}
            {activeReplyId}
            onReplyToggle={(id) => (activeReplyId = id)}
            onReplySubmit={addReply}
          />
          {#if i < comments.length - 1}
            <div class="mt-6 border-t border-[rgba(197,247,208,0.4)]"></div>
          {/if}
        </div>
      {/each}
    </div>
  {/if}

  <!-- Leave a comment form -->
  <div class="mt-8 border-t border-[rgba(197,247,208,0.4)] pt-6">
    <p
      class="font-manrope mb-3 text-[11px] font-bold tracking-[0.12em] text-[rgba(41,48,45,0.5)] uppercase"
    >
      Leave a comment
    </p>
    <textarea
      bind:value={newCommentBody}
      placeholder="What's on your mind?"
      rows="4"
      class="font-manrope w-full resize-none rounded-[8px] border-[1.5px] border-[rgba(197,247,208,0.8)] bg-white px-3.5 py-3 text-[15px] text-[#29302d] placeholder-[rgba(41,48,45,0.35)] outline-none focus:border-[#386547]"
    ></textarea>
    <div class="mt-2.5">
      <button
        type="button"
        onclick={addComment}
        disabled={!newCommentBody.trim()}
        class="font-manrope cursor-pointer rounded-full bg-[#386547] px-5 py-2 text-[14px] font-bold text-white disabled:cursor-not-allowed disabled:opacity-40"
      >
        Post comment
      </button>
    </div>
  </div>
</div>
```
